26  /**
27   * Test case for {@link AjcMojo}.
28   *
29   * @since 0.1
30   */
31  final class AjcMojoTest {
32  
33      @Test
34      @Disabled
35      void weavesClassFiles(@TempDir final Path temp) throws Exception {
36          final MavenProject project = Mockito.mock(MavenProject.class);
37          Mockito.doReturn(Collections.emptyList())
38              .when(project).getCompileClasspathElements();
39          final Path classes = temp.resolve("classes");
40          final Path javas = temp.resolve("javas");
41          final MavenSession session = Mockito.mock(
42              MavenSession.class, Mockito.RETURNS_DEEP_STUBS
43          );
44          Mockito.when(session.getLocalRepository().getBasedir())
45              .thenReturn(temp.resolve("xx").toFile().toString());
46          final Path java = javas.resolve("sample/Foo.java");
47          java.getParent().toFile().mkdirs();
48          Files.write(
49              java,
50              String.join(
51                  System.lineSeparator(),
52                  "package sample;",
53                  "import com.jcabi.aspects.Immutable;",
54                  "@Immutable class Foo {}"
55              ).getBytes(StandardCharsets.UTF_8)
56          );
57          final JavaCompiler compiler = ToolProvider.getSystemJavaCompiler();
58          try (
59              StandardJavaFileManager mgr = compiler.getStandardFileManager(
60                  null, Locale.ENGLISH, StandardCharsets.UTF_8
61              )
62          ) {
63              compiler.getTask(
64                  null, mgr, null, null, null,
65                  mgr.getJavaFileObjectsFromFiles(
66                      Collections.singleton(java.toFile())
67                  )
68              ).call();
69          }
70          final String name = "sample/Foo.class";
71          final Path binary = classes.resolve(name);
72          FileUtils.copyFile(javas.resolve(name).toFile(), binary.toFile());
73          final long size = binary.toFile().length();
74          new Mojo<>(AjcMojo.class)
75              .with("project", project)
76              .with("session", session)
77              .with("classesDirectory", classes.toFile())
78              .with("aspectDirectories", new File[0])
79              .with("tempDirectory", temp.resolve("temps").toFile())
80              .execute();
81          MatcherAssert.assertThat(
82              String.format("weaving cannot leave the size at %d", size),
83              binary.toFile().length(),
84              Matchers.not(Matchers.equalTo(size))
85          );
86      }
87  }
