5   package com.jcabi.maven.plugin;
6   
7   import java.nio.charset.StandardCharsets;
8   import java.nio.file.Files;
9   import java.nio.file.Path;
10  import org.apache.maven.project.MavenProject;
11  import org.hamcrest.MatcherAssert;
12  import org.hamcrest.Matchers;
13  import org.junit.jupiter.api.Test;
14  import org.junit.jupiter.api.io.TempDir;
15  import org.mockito.Mockito;
16  
17  /**
18   * Test case for {@link VersionalizeMojo}.
19   *
20   * @since 0.1
21   */
22  final class VersionalizeMojoTest {
23  
24      @Test
25      void preservesMixedCaseCoordinates(@TempDir final Path temp) throws Exception {
26          final Path sources = Files.createDirectories(temp.resolve("sources"));
27          Files.write(
28              sources.resolve("App.java"),
29              "class App { }".getBytes(StandardCharsets.UTF_8)
30          );
31          final Path classes = temp.resolve("classes");
32          final MavenProject project = Mockito.mock(
33              MavenProject.class, Mockito.RETURNS_DEEP_STUBS
34          );
35          Mockito.when(project.getBuild().getSourceDirectory())
36              .thenReturn(sources.toString());
37          Mockito.when(project.getBuild().getOutputDirectory())
38              .thenReturn(classes.toString());
39          Mockito.when(project.getGroupId()).thenReturn("com.Example");
40          Mockito.when(project.getArtifactId()).thenReturn("MyApp");
41          Mockito.when(project.getPackaging()).thenReturn("jar");
42          new Mojo<>(VersionalizeMojo.class)
43              .with("project", project)
44              .execute();
45          MatcherAssert.assertThat(
46              "version filename must preserve uppercase coordinate letters",
47              classes.resolve("com-Example-MyApp-jar.txt").toFile().exists(),
48              Matchers.is(true)
49          );
50      }
51  
52      @Test
53      void skipsExecutionWhenRequired(@TempDir final Path temp) {
54          final MavenProject project = Mockito.mock(
55              MavenProject.class, Mockito.RETURNS_DEEP_STUBS
56          );
57          Mockito.when(project.getBuild().getSourceDirectory())
58              .thenReturn(temp.resolve("sources").toString());
59          Mockito.when(project.getBuild().getOutputDirectory())
60              .thenReturn(temp.resolve("classes").toString());
61          new Mojo<>(VersionalizeMojo.class)
62              .with("project", project)
63              .execute();
64          MatcherAssert.assertThat(
65              "output directory cannot appear when sources are absent",
66              temp.resolve("classes").toFile().exists(),
67              Matchers.is(false)
68          );
69      }
70  }
