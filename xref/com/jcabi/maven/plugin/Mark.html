5   package com.jcabi.maven.plugin;
6   
7   import java.io.File;
8   import java.io.IOException;
9   import java.nio.file.Files;
10  import java.util.Arrays;
11  import java.util.Collection;
12  import org.apache.commons.io.FileUtils;
13  import org.apache.commons.io.filefilter.TrueFileFilter;
14  
15  /**
16   * Marker file that remembers the moment of the last successful weave,
17   * so that classes already woven are never handed to AspectJ again.
18   *
19   * @since 0.18
20   */
21  final class Mark {
22  
23      /**
24       * The marker file.
25       */
26      private final transient File file;
27  
28      /**
29       * Directories whose files are compared with the marker.
30       */
31      private final transient File[] dirs;
32  
33      /**
34       * Ctor.
35       *
36       * @param marker The marker file
37       * @param watched Directories whose files are compared with the marker
38       */
39      Mark(final File marker, final File... watched) {
40          this.file = marker;
41          this.dirs = watched.clone();
42      }
43  
44      /**
45       * Is there a file newer than the marker, or no marker at all?
46       *
47       * @return TRUE if weaving is required
48       */
49      boolean stale() {
50          return !this.file.exists() || Arrays.stream(this.dirs)
51              .filter(File::isDirectory)
52              .map(Mark::files)
53              .flatMap(Collection::stream)
54              .anyMatch(f -> FileUtils.isFileNewer(f, this.file));
55      }
56  
57      /**
58       * Create the marker or update its modification time.
59       *
60       * @throws IOException If fails
61       */
62      void touch() throws IOException {
63          FileUtils.touch(this.file);
64      }
65  
66      /**
67       * Delete the marker, if it exists.
68       *
69       * @throws IOException If fails
70       */
71      void erase() throws IOException {
72          Files.deleteIfExists(this.file.toPath());
73      }
74  
75      private static Collection<File> files(final File dir) {
76          return FileUtils.listFiles(
77              dir, TrueFileFilter.INSTANCE, TrueFileFilter.INSTANCE
78          );
79      }
80  }
